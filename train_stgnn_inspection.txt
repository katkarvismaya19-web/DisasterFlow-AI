325: # ============================================================
326: # GRAPH CONVOLUTION
327: # ============================================================
328: 
329: class GraphConv(nn.Module):
330: 
331:     def __init__(
332:         self,
333:         in_features,
334:         out_features
335:     ):
336: 
337:         super().__init__()
338: 
339:         self.linear = nn.Linear(
340:             in_features,
341:             out_features
342:         )
343: 
344:     def forward(
345:         self,
346:         x,
347:         adjacency
348:     ):
349: 
350:         # x:
351:         # B × N × F
352: 
353:         x = torch.einsum(
354:             "ij,bjf->bif",
355:             adjacency,
356:             x
357:         )
358: 
359:         return self.linear(x)

362: # ============================================================
363: # ST-GNN
364: # ============================================================
365: 
366: class STGNN(nn.Module):
367: 
368:     def __init__(
369:         self,
370:         input_features,
371:         hidden_size,
372:         graph_hidden,
373:         nodes
374:     ):
375: 
376:         super().__init__()
377: 
378:         self.nodes = nodes
379: 
380:         # Temporal encoder
381:         self.temporal = nn.GRU(
382:             input_size=input_features,
383:             hidden_size=hidden_size,
384:             batch_first=True
385:         )
386: 
387:         # Spatial graph layers
388:         self.graph1 = GraphConv(
389:             hidden_size,
390:             graph_hidden
391:         )
392: 
393:         self.graph2 = GraphConv(
394:             graph_hidden,
395:             graph_hidden
396:         )
397: 
398:         # Attention over node representation
399:         self.node_attention = nn.Sequential(
400:             nn.Linear(
401:                 graph_hidden,
402:                 32
403:             ),
404:             nn.ReLU(),
405:             nn.Linear(
406:                 32,
407:                 1
408:             )
409:         )
410: 
411:         # Final node-level predictor
412:         self.classifier = nn.Sequential(
413:             nn.Linear(
414:                 graph_hidden,
415:                 32
416:             ),
417:             nn.ReLU(),
418:             nn.Dropout(0.2),
419:             nn.Linear(
420:                 32,
421:                 1
422:             )
423:         )
424: 
425:     def forward(
426:         self,
427:         x,
428:         adjacency
429:     ):
430: 
431:         # ----------------------------------------------------
432:         # x = B × T × N × F
433:         # ----------------------------------------------------
434: 
435:         batch_size = x.size(0)
436: 
437:         # Rearrange:
438:         # B × N × T × F
439: 
440:         x = x.permute(
441:             0,
442:             2,
443:             1,
444:             3
445:         )
446: 
447:         # B*N × T × F
448: 
449:         x = x.reshape(
450:             batch_size * self.nodes,
451:             x.size(2),
452:             x.size(3)
453:         )
454: 
455:         # Temporal representation
456: 
457:         temporal_output, _ = self.temporal(x)
458: 
459:         temporal_last = (
460:             temporal_output[:, -1, :]
461:         )
462: 
463:         # B × N × H
464: 
465:         node_features = temporal_last.reshape(
466:             batch_size,
467:             self.nodes,
468:             -1
469:         )
470: 
471:         # Spatial graph propagation
472: 
473:         node_features = self.graph1(
474:             node_features,
475:             adjacency
476:         )
477: 
478:         node_features = torch.relu(
479:             node_features
480:         )
481: 
482:         node_features = self.graph2(
483:             node_features,
484:             adjacency
485:         )
486: 
487:         node_features = torch.relu(
488:             node_features
489:         )
490: 
491:         # Attention weighting
492: 
493:         attention = torch.softmax(
494:             self.node_attention(
495:                 node_features
496:             ).squeeze(-1),
497:             dim=1
498:         )
499: 
500:         node_features = (
501:             node_features
502:             *
503:             (1.0 + attention.unsqueeze(-1))
504:         )
505: 
506:         # Node-level flood prediction
507: 
508:         logits = self.classifier(
509:             node_features
510:         ).squeeze(-1)
511: 
512:         return logits
513:
